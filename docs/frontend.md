```tsx
    i18n.changeLanguage(i18n.language === 'th' ? 'en' : 'th');
  };

  return (
    <div>
      <h1>{t('products.title')}</h1>
      <button onClick={toggleLang}>{i18n.language === 'th' ? 'EN' : 'ไทย'}</button>
```

### 5.4 Rules

- ห้าม hardcode user-facing strings — ใช้ `t()` เสมอ
- Translation key ต้อง nested ตาม feature: `products.fields.name`
- Pluralization ใช้ i18next plural keys
- Date/number formatting ใช้ `Intl.DateTimeFormat` / `Intl.NumberFormat` กับ `i18n.language`

## 6. API Client

**`src/api/client.ts`:**

```typescript
import axios, { AxiosError, type InternalAxiosRequestConfig } from 'axios';

import { useAuthStore } from '@/stores/authStore';

export const apiClient = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || '/api',
  timeout: 15_000,
  headers: { 'Content-Type': 'application/json' },
});

apiClient.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    const token = useAuthStore.getState().accessToken;
    if (token) {
      config.headers.set('Authorization', `Bearer ${token}`);
    }
    config.headers.set('X-Request-Id', crypto.randomUUID());
    return config;
  },
);

apiClient.interceptors.response.use(
  (response) => response,
  async (error: AxiosError) => {
    if (error.response?.status === 401) {
      useAuthStore.getState().clearAuth();
      window.location.href = '/login';
    }
    return Promise.reject(error);
  },
);

export interface ApiError {
  code: string;
  message: string;
  details?: { field: string; code: string; message: string }[];
  requestId: string;
}

export function extractApiError(error: unknown): ApiError {
  if (axios.isAxiosError(error) && error.response?.data?.error) {
    return error.response.data.error as ApiError;
  }
  return {
    code: 'UNKNOWN_ERROR',
    message: error instanceof Error ? error.message : 'Unknown error',
    requestId: 'n/a',
  };
}
```

**`src/api/products.ts`:**

```typescript
import { apiClient } from './client';

export interface Product {
  id: string;
  name: string;
  sku: string;
  businessUnitId: string;
  category: string;
  price: string;
  status: 'draft' | 'active' | 'discontinued';
  createdAt: string;
  updatedAt: string;
}

export interface CursorPage<T> {
  data: T[];
  pagination: { nextCursor: string | null; hasMore: boolean };
}

export interface ProductListParams {
  cursor?: string;
  limit?: number;
  status?: string;
}

export async function listProducts(params: ProductListParams = {}): Promise<CursorPage<Product>> {
  const { data } = await apiClient.get<CursorPage<Product>>('/products', { params });
  return data;
}

export async function getProduct(id: string): Promise<Product> {
  const { data } = await apiClient.get<Product>(`/products/${id}`);
  return data;
}

export interface ProductCreatePayload {
  name: string;
  sku: string;
  businessUnitId: string;
  category: string;
  price: string;
  status: Product['status'];
}

export async function createProduct(payload: ProductCreatePayload): Promise<Product> {
  const { data } = await apiClient.post<Product>('/products', payload);
  return data;
}
```

```tsx
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { listProducts, createProduct } from '@/api/products';

export function useProducts(params: ProductListParams = {}) {
  return useQuery({
    queryKey: ['products', params],
    queryFn: () => listProducts(params),
  });
}

export function useCreateProduct() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: createProduct,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['products'] });
    },
  });
}
```

### 7.2 Client State → Zustand

**`src/stores/authStore.ts`:**

```typescript
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface AuthState {
  accessToken: string | null;
  user: { id: string; email: string; roles: string[] } | null;
  setAuth: (token: string, user: AuthState['user']) => void;
  clearAuth: () => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      accessToken: null,
      user: null,
      setAuth: (accessToken, user) => set({ accessToken, user }),
      clearAuth: () => set({ accessToken: null, user: null }),
    }),
    {
      name: 'auth-storage',
      partialize: (state) => ({ accessToken: state.accessToken, user: state.user }),
    },
  ),
);
```

**Rule of thumb:**
- Server data (fetched from API) → **React Query**
- UI state (modals, filters, form steps) → **useState** ใน component
- Cross-component client state (auth, theme, current org) → **Zustand**

## 8. Forms (React Hook Form + Zod)

```tsx
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';
import { useTranslation } from 'react-i18next';

const productSchema = z.object({
  name: z.string().min(1).max(200),
  sku: z.string().regex(/^[A-Z0-9-]{3,32}$/),
  businessUnitId: z.string().uuid(),
  category: z.string().max(50),
  price: z.string().regex(/^\d+(\.\d{1,2})?$/),
  status: z.enum(['draft', 'active', 'discontinued']),
});

type ProductFormValues = z.infer<typeof productSchema>;

export function ProductForm({ onSubmit }: { onSubmit: (values: ProductFormValues) => void }) {
  const { t } = useTranslation();
  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting },
  } = useForm<ProductFormValues>({
    resolver: zodResolver(productSchema),
    defaultValues: { status: 'draft' },
  });

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <label>
        {t('products.fields.name')}
        <input type="text" {...register('name')} />
        {errors.name && <span>{errors.name.message}</span>}
      </label>

      <label>
        {t('products.fields.sku')}
        <input type="text" {...register('sku')} />
        {errors.sku && <span>{errors.sku.message}</span>}
      </label>

      <label>
        {t('products.fields.price')}
        <input type="text" {...register('price')} />
        {errors.price && <span>{errors.price.message}</span>}
      </label>

      <button type="submit" disabled={isSubmitting}>
        {isSubmitting ? t('common.loading') : t('common.save')}
      </button>
    </form>
  );
}
```

## 9. Error Boundary

**`src/components/ErrorBoundary.tsx`:**

```tsx
import { Component, type ErrorInfo, type ReactNode } from 'react';

interface Props {
  children: ReactNode;
  fallback?: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

export class ErrorBoundary extends Component<Props, State> {
  state: State = { hasError: false, error: null };

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo): void {
    console.error('ErrorBoundary caught:', error, errorInfo);
  }

  render(): ReactNode {
    if (this.state.hasError) {
      return (
        this.props.fallback ?? (
          <div role="alert">
            <h1>Something went wrong</h1>
            <p>{this.state.error?.message}</p>
            <button onClick={() => window.location.reload()}>Reload</button>
          </div>
        )
      );
    }
    return this.props.children;
  }
}
```

## 12. Standard Layout (AppLayout) — L1 Baseline

ตั้งแต่ auth absorption sprint — standard ติด layout shell ครบในทุก project ที่ scaffold ใหม่
ไฟล์อยู่ใน `frontend/src/components/Layout/`:

```
Layout/
├── AppLayout.tsx     # shell: <div>TopBar / Sidebar / <Outlet/></div>
├── TopBar.tsx        # project_display_name (ซ้าย) + UserMenu (ขวา)
├── Sidebar.tsx       # อ่าน /api/v1/me/menus → render tree, server-filtered
└── UserMenu.tsx      # avatar dropdown: Profile, "ออกจากระบบ"
```

**App shell** — `AppLayout` is a fixed-height shell (`h-dvh … overflow-hidden`):
the viewport never scrolls as a whole. `TopBar` is a fixed-height row at top; below
it a `flex min-h-0 flex-1` row holds `Sidebar` + `main`, each scrolling
independently. The `min-h-0` on that row is load-bearing — without it flex children
refuse to shrink and overflow is **clipped** instead of scrolling (the classic
"menu cut off at the last visible item" bug).

**Sidebar behavior** — the `aside` is a flex column; its `nav` is
`flex-1 min-h-0 overflow-y-auto`, so long menu trees scroll **inside** the sidebar
on both the desktop rail and the mobile drawer, and every item stays reachable.
No magic `calc()` height — the sidebar fills the shell row via flex, so it stays
correct even if `TopBar`'s height changes.

**Main behavior** — `main` is `flex-1 overflow-y-auto`; page content scrolls there,
not on the body, so `TopBar` and `Sidebar` stay pinned without needing `sticky`.

**App.tsx pattern** (emitted by scaffold):

```tsx
import { AuthBootstrap } from './components/AuthBootstrap';
import { RequireAuth } from './components/RequireAuth';
import { RequirePermission } from './components/RequirePermission';
import { AppLayout } from './components/Layout/AppLayout';

<AuthBootstrap>
  <Routes>
    <Route path="/login" element={<Login />} />
    <Route element={<RequireAuth><AppLayout /></RequireAuth>}>
      <Route path="/dashboard" element={<Dashboard />} />
      <Route path="/settings" element={<SettingsIndex />} />
      <Route
        path="/settings/users"
        element={
          <RequirePermission perm="menu.settings.users.view">
            <SettingsUsers />
          </RequirePermission>
        }
      />
      {/* ... settings/roles, /menus, /auth all wrapped the same way */}
    </Route>
  </Routes>
</AuthBootstrap>
```

**Sidebar data source** — `useQuery(['me-menus'], () => GET /api/v1/me/menus)` — server filters
the menu tree by current user's effective permissions (role + per-user overrides) before
returning, so the frontend doesn't need to know which `menu.*.view` perm gates which path.

**RequireAuth contract** — เช็ค `useAuthStore.getState().user`; ถ้า null → `<Navigate to="/login?return=..." replace />`
(preserves the originally-requested path so post-login bounces back there)

**RequirePermission contract** — wraps a single route element. Reads
`useHasPermission(perm)` → ถ้า false render 403 fallback (ไม่ redirect — user is authed,
just missing perm). Defense-in-depth on top of the server-side check.

### 12.1 Zustand Auth Store

**ไฟล์:** `frontend/src/stores/auth.ts`

```typescript
import { create } from 'zustand';

interface AuthState {
  user: User | null;
  permissionKeys: string[];
  isAuthenticated: boolean;
  hydrateFromServer: () => Promise<void>;
  login: (email: string, password: string) => Promise<void>;
```

**กฎเหล็ก:**
- **In-memory only** — ไม่ persist user/permissionKeys ลง localStorage (XSS exfil vector)
- Access token เก็บใน `sessionStorage` ผ่าน `lib/auth-token.ts` (ห้ามใช้ localStorage โดยตรง — ดู Bug 19)
- Refresh token เป็น httpOnly cookie ที่ backend set (ไม่ touch จาก JS)
- `hydrateFromServer()` เรียก `GET /api/v1/me` + `/me/permissions` หลัง silent refresh สำเร็จ
  (ใน `AuthBootstrap`) เพื่อรื้อ state กลับมาหลัง page reload

### 12.2 Login Page — Fail-Open ต่อ /admin/settings/public 500

`frontend/src/pages/Login.tsx` ขอ `getPublicAuthSettings()` ก่อนตัดสินใจว่าจะ render
local form / SSO button / ทั้งคู่ — แต่ **ถ้า endpoint นั้น error** (500 / network down)
หน้า Login ไม่ block — fall back ไปแสดงทั้งสอง provider พร้อมกัน:

```tsx
const localEnabled = settingsError ? true : settings!.authLocalEnabled;
const ssoEnabled   = settingsError ? true : settings!.authSsoEnabled;
```

Rationale: backend ยังคงบังคับ correct toggle อยู่ — frontend แค่ guess UI ผิด user
จะกดแล้ว 403 (ดีกว่า login page hang เพราะ DB issue)

## 13. Quick Reference: เมื่อ AI ได้รับ task

1. **Read first:** AGENTS.md + this file
2. **Add new feature:**
   - Add API function in `src/api/<resource>.ts`
   - Add React Query hooks in `src/features/<feature>/hooks.ts`
   - Add components in `src/features/<feature>/`
   - Add route in `App.tsx`
   - Add translations in `th.json` and `en.json` (ทั้งสองภาษาเสมอ)
3. **Forms:** ใช้ react-hook-form + zod เสมอ ห้ามใช้ uncontrolled forms
4. **No hardcoded strings:** ใช้ `t()` ทุก user-facing text
5. **Update handover docs** ถ้ามีการเปลี่ยน env vars หรือ external dependencies
